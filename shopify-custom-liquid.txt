<script>
(() => {
  const BUILDER_URL = 'YOUR_GITHUB_PAGES_URL';
  const EXPECTED_VARIANT_ID = '57150364811588';
  let builderOrigin = '';

  try {
    builderOrigin = new URL(BUILDER_URL).origin;
  } catch (e) {
    console.error('Achilles Kit Builder: replace YOUR_GITHUB_PAGES_URL in the Custom Liquid code.');
    return;
  }

  window.addEventListener('message', async (event) => {
    if (event.origin !== builderOrigin) return;
    const data = event.data || {};
    if (data.type !== 'ACHILLES_ADD_TO_CART') return;

    const reply = (payload) => {
      try { event.source.postMessage(payload, event.origin); } catch (e) {}
    };

    // Do not trust a product ID supplied by a cross-origin iframe.
    if (String(data.variantId) !== EXPECTED_VARIANT_ID) {
      reply({ type: 'ACHILLES_CART_RESULT', ok: false, message: 'The kit product configuration does not match Shopify.' });
      return;
    }

    try {
      const formData = new FormData();
      formData.append('id', EXPECTED_VARIANT_ID);
      formData.append('quantity', String(Math.max(1, Number(data.quantity) || 1)));

      const properties = data.properties || {};
      Object.entries(properties).forEach(([key, value]) => {
        formData.append(`properties[${key}]`, String(value ?? ''));
      });

      // File objects can be structured-cloned through postMessage in modern browsers.
      // If Shopify rejects a particular file type/size, the textual design still remains intact.
      const artwork = Array.isArray(data.artwork) ? data.artwork : [];
      artwork.forEach((item) => {
        if (item && item.label && item.file instanceof File) {
          formData.append(`properties[${item.label}]`, item.file, item.file.name);
        }
      });

      const root = (window.Shopify && window.Shopify.routes && window.Shopify.routes.root) || '/';
      const response = await fetch(root + 'cart/add.js', {
        method: 'POST',
        body: formData,
        credentials: 'same-origin'
      });

      const result = await response.json().catch(() => ({}));
      if (!response.ok) {
        throw new Error(result.description || result.message || 'Shopify could not add this kit to the basket.');
      }

      reply({ type: 'ACHILLES_CART_RESULT', ok: true });
      window.location.assign(root + 'cart');
    } catch (error) {
      console.error('Achilles Kit Builder cart error:', error);
      reply({ type: 'ACHILLES_CART_RESULT', ok: false, message: error.message || 'Could not add the kit to your basket.' });
    }
  });
})();
</script>
